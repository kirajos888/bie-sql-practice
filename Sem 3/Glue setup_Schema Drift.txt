# Reading with DynamicFrame and cleaning Schema Drift

from awsglue.dynamicframe import DynamicFrame
from awsglue.transforms import ResolvedChoice

# 1. Read data directly from the Glue data Catalog

datasource0 = glueContext.create_dynamic_frame.from_catalog(
	database = "analytics_db",
	table_name = "raw_json_logs",
	transformations_ctx = datasource0, # Tells bookmark: "Start tracking files from this step"
)

# 2. Fix Schema Drift (cast all user_id values to string)

resolved_df = ResolvedChoice.apply(
	frame = datasource0,
	choice = "cast:string", # Forces all conflicting types into strings without losing data
	transformations_ctx = "resolved_df',
)

# 3. Write the cleaned data to S3 as Parquet partitioning into year/month subfolders

glueContext.write_dynamic_frame.from_options(
	frame = resolved_df,
	connection_type = "s3",
	connection_options = {
		"path" : "s3://my-datalake-bucket/curated/",
		"partitionKeys": ["year", "month"], # Creates /year=2010/month=10/ subfolders
	}
	format = "parquet",
	transformations_ctx = "write_s3",
)
